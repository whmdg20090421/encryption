================================
四、PropID 枚举（属性标识）
================================

常用属性:
  PATH              String   文件完整路径（如 "dir/file.ext"）
  IS_FOLDER         Boolean  是否为目录
  SIZE              Long     原始大小（未压缩）
  PACKED_SIZE       Long     压缩后大小
  ENCRYPTED         Boolean  是否加密
  ATTRIBUTES        Integer  文件属性位掩码
  LAST_MODIFICATION_TIME  Date  最后修改时间
  CREATION_TIME     Date     创建时间
  LAST_ACCESS_TIME  Date     最后访问时间

属性值类型:
  getProperty() 返回 Object，需根据 PropID 转型:
    PropID.IS_FOLDER → Boolean
    PropID.SIZE → Long
    PropID.ENCRYPTED → Boolean
    PropID.PATH → String
    PropID.PACKED_SIZE → Long

================================
十九、常用使用模式
================================

--- 1. 列出压缩包内容 ---
  val raf = RandomAccessFileInStream(RandomAccessFile(file, "r"))
  val inArchive = SevenZip.openInArchive(null, raf, "password")
  val count = inArchive.numberOfItems
  for (i in 0 until count) {
      val path = inArchive.getStringProperty(i, PropID.PATH)
      val isDir = inArchive.getProperty(i, PropID.IS_FOLDER) as Boolean
      val size = inArchive.getProperty(i, PropID.SIZE) as Long
  }
  inArchive.close()

--- 2. 提取全部文件 ---
  inArchive.extract(null, false, object : IArchiveExtractCallback {
      override fun getStream(index: Int, mode: ExtractAskMode): ISequentialOutStream {
          return ISequentialOutStream { data ->
              // 写入文件
              data.size
          }
      }
      override fun prepareOperation(mode: ExtractAskMode) {}
      override fun setOperationResult(result: ExtractOperationResult) {}
      override fun setTotal(total: Long) {}
      override fun setCompleted(complete: Long) {}
  })

--- 3. 提取单个文件 ---
  inArchive.extract(intArrayOf(index), false, callback)

--- 4. 创建 7z 压缩包 ---
  val outArchive = SevenZip.openOutArchive7z()
  outArchive.setLevel(5)
  outArchive.setHeaderEncryption(true)  // 可选
  outArchive.createArchive(outStream, items.size, callback)
  outArchive.close()

--- 5. 创建 Zip 压缩包 ---
  val outArchive = SevenZip.openOutArchiveZip()
  outArchive.setLevel(5)
  outArchive.createArchive(outStream as IOutStream, items.size, callback)
  outArchive.close()

--- 6. 创建 Tar 压缩包 ---
  val outArchive = SevenZip.openOutArchiveTar()
  outArchive.createArchive(outStream, items.size, callback)
  outArchive.close()
